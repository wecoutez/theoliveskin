function safe(v) { // 시트 수식으로 읽히지 않게
  v = String(v == null ? '' : v).trim().slice(0, 500);
  return /^[=+\-@]/.test(v) ? "'" + v : v;
}
function fmtDate(v) { return v instanceof Date ? Utilities.formatDate(v, TZ, 'yyyy-MM-dd') : v; }

/* ---------- 제목(1행)으로 열을 찾는 표 ---------- */
function table(def) {
  let sh = null;
  for (const n of def.names) { sh = book().getSheetByName(n); if (sh) break; }
  if (!sh) sh = book().insertSheet(def.names[0]);
  const head = sh.getLastRow() >= 1 && sh.getLastColumn() >= 1
    ? sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0].map(h => String(h).trim()) : [];
  const col = {};
  def.fields.forEach(([k, label]) => {
    let i = head.findIndex(h => h === k || h === label);
    if (i < 0) {
      i = head.length; head.push(label);
      sh.getRange(1, i + 1).setValue(label).setFontWeight('bold');
    }
    col[k] = i;
    if (!def.nums.includes(k)) sh.getRange(1, i + 1, sh.getMaxRows(), 1).setNumberFormat('@'); // 날짜·전화번호를 글자 그대로
  });
  if (sh.getFrozenRows() < 1) sh.setFrozenRows(1);
  return { sh, col, width: head.length };
}
function readRows(def) {
  const t = table(def);
  if (t.sh.getLastRow() < 2) return [];
  const vals = t.sh.getRange(2, 1, t.sh.getLastRow() - 1, Math.max(t.width, t.sh.getLastColumn())).getValues();
  return vals.filter(r => String(r[t.col.id]).trim()).map(r => {
    const o = {};
    def.fields.forEach(([k]) => {
      const v = fmtDate(r[t.col[k]]);
      o[k] = def.nums.includes(k) ? (Number(v) || 0) : String(v == null ? '' : v);
    });
    return o;
  });
}
function findRow(t, id) {
  if (t.sh.getLastRow() < 2) return 0;
  const ids = t.sh.getRange(2, t.col.id + 1, t.sh.getLastRow() - 1, 1).getValues();
  for (let i = 0; i < ids.length; i++) if (String(ids[i][0]) === String(id)) return i + 2;
  return 0;
}
function saveRow(def, o) {
  const t = table(def);
  const id = o.id ? String(o.id) : Utilities.getUuid();
  const r = findRow(t, id);
  const width = Math.max(t.width, t.sh.getLastColumn());
  const row = r ? t.sh.getRange(r, 1, 1, width).getValues()[0] : new Array(width).fill(''); // 표에 없는 열은 건드리지 않음
  def.fields.forEach(([k]) => {
    if (k === 'id') row[t.col.id] = id;
    else if (k === 'createdAt') row[t.col.createdAt] = Number(o.createdAt) || (r ? row[t.col.createdAt] : 0) || Date.now();
    else if (def.nums.includes(k)) row[t.col[k]] = Number(o[k]) || 0;
    else if (k in o) row[t.col[k]] = safe(o[k]);
  });
  if (r) t.sh.getRange(r, 1, 1, width).setValues([row]); else t.sh.appendRow(row);
}
function deleteRow(def, id) {
  const t = table(def); const r = findRow(t, id);
  if (r) t.sh.deleteRow(r);
}

function settingSheet() {
  let sh = book().getSheetByName(SETTING_SHEET);
  if (!sh) {
    sh = book().insertSheet(SETTING_SHEET);
    sh.getRange(1, 1, 1, 2).setValues([['항목', '값']]).setFontWeight('bold');
    sh.setFrozenRows(1);
  }
  return sh;
}
function readSettings() {
  const st = settingSheet(), s = {};
  if (st.getLastRow() > 1) st.getRange(2, 1, st.getLastRow() - 1, 2).getValues().forEach(([k, v]) => {
    k = String(k).trim();
    if (!SETTING_KEYS.includes(k)) return;
    v = fmtDate(v);
    s[k] = k === 'deadline' ? String(v || '') : Number(v) || 0;
  });
  return s;
}
function saveSettings(s) {
  const st = settingSheet();
  const keys = st.getLastRow() > 1 ? st.getRange(2, 1, st.getLastRow() - 1, 1).getValues().map(r => String(r[0]).trim()) : [];
  SETTING_KEYS.forEach(k => {
    if (!(k in s)) return;
    const v = k === 'deadline' ? String(s[k] || '') : Number(s[k]) || 0;
    let i = keys.indexOf(k);
    if (i < 0) { keys.push(k); i = keys.length - 1; }
    st.getRange(i + 2, 2).setNumberFormat('@');
    st.getRange(i + 2, 1, 1, 2).setValues([[k, v]]);
  });
}

function mktSheet() {
  let sh = book().getSheetByName(MKT_SHEET);
  if (!sh) {
    sh = book().insertSheet(MKT_SHEET);
    sh.getRange(1, 1, 1, 2).setValues([['part', 'json']]).setFontWeight('bold');
    sh.setFrozenRows(1);
  }
  return sh;
}
function mktRead() {
  const sh = mktSheet();
  if (sh.getLastRow() < 2) return null;
  const txt = sh.getRange(2, 2, sh.getLastRow() - 1, 1).getValues().map(r => String(r[0]).replace(/^~/, '')).join('');
  if (!txt) return null;
  try { return JSON.parse(txt); } catch (e) { return null; }
}
function mktWrite(m) {
  const txt = JSON.stringify(m || {});
  if (txt.length > 2000000) throw new Error('too_big');
  const sh = mktSheet();
  if (sh.getLastRow() > 1) sh.getRange(2, 1, sh.getLastRow() - 1, 2).clearContent();
  const rows = [];
  for (let i = 0; i < txt.length; i += MKT_CHUNK) rows.push([rows.length + 1, '~' + txt.slice(i, i + MKT_CHUNK)]); // ~ : 수식으로 읽히지 않게
  sh.getRange(2, 2, rows.length, 1).setNumberFormat('@');
  sh.getRange(2, 1, rows.length, 2).setValues(rows);
}

function amountFor(cc, qty) {
  const ship = shipFor(cc, qty);

}

// 위 ▼ 설정 칸 값을 먼저 쓰고, 비어 있으면 프로젝트 설정의 스크립트 속성을 씀 (예전 방식도 그대로 됨)
function prop(k) { const v = String(SETUP[k] || '').trim(); return v || PropertiesService.getScriptProperties().getProperty(k); }
function ppBase() { return prop('PAYPAL_ENV') === 'sandbox' ? 'https://api-m.sandbox.paypal.com' : 'https://api-m.paypal.com'; }
function ppToken() {
  const cache = CacheService.getScriptCache();
  const hit = cache.get('pp_token');
  if (hit) return hit;
  const r = UrlFetchApp.fetch(ppBase() + '/v1/oauth2/token', {
    method: 'post',
    headers: { Authorization: 'Basic ' + Utilities.base64Encode(prop('PAYPAL_CLIENT_ID') + ':' + prop('PAYPAL_SECRET')) },
    payload: { grant_type: 'client_credentials' },
    muteHttpExceptions: true
  });
  const j = JSON.parse(r.getContentText());
  if (!j.access_token) throw new Error('paypal_auth ' + r.getContentText().slice(0, 200));
  cache.put('pp_token', j.access_token, Math.min(21600, (j.expires_in || 3600) - 300));
  return j.access_token;
}
function pp(method, path, body, extraHeaders) {
  const opt = {
    method: method,
    contentType: 'application/json',
    headers: Object.assign({ Authorization: 'Bearer ' + ppToken() }, extraHeaders || {}),
    muteHttpExceptions: true
  };
  if (body !== undefined) opt.payload = JSON.stringify(body);
  const r = UrlFetchApp.fetch(ppBase() + path, opt);
  const t = r.getContentText();
  return { code: r.getResponseCode(), body: t ? JSON.parse(t) : {} };
}
function issue(res) { return (res.body && res.body.details && res.body.details[0] && res.body.details[0].issue) || ''; }

function createOrder(b) {
  const qty = Math.max(1, Math.min(MAX_QTY, parseInt(b.qty, 10) || 1));
  const cc = String(b.country || 'US').toUpperCase();
  const amount = amountFor(cc, qty);
  if (!amount) return { ok: false, error: 'no_ship' };
  const res = pp('post', '/v2/checkout/orders', {
    intent: 'CAPTURE',
    purchase_units: [{
      reference_id: 'default',
      custom_id: cc + '|' + qty,           // 주문 만들 때 기준 국가·수량 (확정 때 비교용)
      description: 'Olive Skin Glow Drop 15ml',
      amount: amount,
      items: [{
        name: 'Glow Drop Daily Care Oil 15ml', sku: SKU,
        unit_amount: { currency_code: 'USD', value: PRICE.toFixed(2) },
        quantity: String(qty), category: 'PHYSICAL_GOODS'
      }]
    }],
    application_context: { brand_name: 'Olive Skin', shipping_preference: 'GET_FROM_FILE' }
  });
  if (res.code >= 300 || !res.body.id) return { ok: false, error: 'create', detail: issue(res) };
  return { ok: true, id: res.body.id };
}

  if (res.code >= 300) return { ok: false, error: 'patch', detail: issue(res) };
  return { ok: true };
}

function captureOrder(b) {
  const id = String(b.orderID || '');
  const got = pp('get', '/v2/checkout/orders/' + encodeURIComponent(id));
  if (got.code >= 300) return { ok: false, error: 'capture' };
  const o = got.body;
  if (o.status === 'COMPLETED') return { ok: true, id: id };   // 이미 확정된 주문 (새로고침 등)
  const pu = o.purchase_units[0];
  const qty = orderQty(o);
  const cc = String((pu.shipping && pu.shipping.address && pu.shipping.address.country_code) || '').toUpperCase();
  // 구매자가 PayPal 에서 입력한 나라 기준으로 금액이 맞는지 확인 — 다르면 확정하지 않음 (청구 안 됨)
  const want = amountFor(cc, qty);
  if (!want) return { ok: false, error: 'no_ship', country: cc };
  if (want.value !== pu.amount.value) return { ok: false, error: 'country_mismatch', country: cc };

  const res = pp('post', '/v2/checkout/orders/' + encodeURIComponent(id) + '/capture', {}, { 'PayPal-Request-Id': 'cap-' + id });
  if (issue(res) === 'INSTRUMENT_DECLINED') return { ok: false, retry: true };
  if (issue(res) === 'ORDER_ALREADY_CAPTURED') return { ok: true, id: id };
  if (res.code >= 300 || res.body.status !== 'COMPLETED') return { ok: false, error: 'capture', detail: issue(res) };
  try { logOrder(res.body, qty, want); } catch (e) { console.error('log failed', e); }  // 기록 실패해도 결제는 완료
  return { ok: true, id: id };
}

function orderQty(o) {
  const it = o.purchase_units && o.purchase_units[0].items && o.purchase_units[0].items[0];
  return Math.max(1, Math.min(MAX_QTY, parseInt(it && it.quantity, 10) || 1));
}

/* ===== 시트 기록 ===== */
function logOrder(o, qty, amount) {
  const ss = book();
  let sh = ss.getSheetByName(ORDER_SHEET);
  if (!sh) {
    sh = ss.insertSheet(ORDER_SHEET);
    sh.appendRow(['결제 시각', 'PayPal 주문번호', '결제번호', '수량', '상품 금액', '배송비', '합계(USD)', '받는 사람', '이메일', '국가', '주소', '진행 상태']);
    sh.getRange(1, 1, 1, 12).setFontWeight('bold'); sh.setFrozenRows(1);
  }
  const pu = o.purchase_units[0];
  const cap = pu.payments && pu.payments.captures && pu.payments.captures[0];
  const a = (pu.shipping && pu.shipping.address) || {};
  const addr = [a.address_line_1, a.address_line_2, a.admin_area_2, a.admin_area_1, a.postal_code].filter(String).join(', ');
  const safe = v => /^[=+\-@]/.test(String(v)) ? "'" + v : v;   // 시트 수식으로 읽히지 않게
  sh.appendRow([
    Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd HH:mm'), o.id, cap ? cap.id : '', qty,
    amount.breakdown.item_total.value, amount.breakdown.shipping.value, amount.value,
    safe((pu.shipping && pu.shipping.name && pu.shipping.name.full_name) || ''),
    safe((o.payer && o.payer.email_address) || ''), a.country_code || '', safe(addr), '발송 대기'
  ]);
}

/* ===== 국내 주문 (페이앱) ===== */
const KR_SHEET = '국내 주문';
const KR_PRICE = 16000;     // 한 병 가격 (원) — kr/checkout.html 과 같게
const KR_MAX = 10;
// 도매 (네일샵·살롱) — kr/checkout.html 과 같게
const B2B = { unit: 7000, bulkUnit: 6000, bulkFrom: 50, step: 10, max: 500 };
const KR_HEAD = ['주문 시각', '주문번호', '상태', '수량', '결제 금액', '이름', '휴대폰', '이메일', '우편번호', '주소', '상세주소', '배송 메모', '페이앱 결제번호', '결제 시각', '실제 결제 금액', '구분', '상호명', '사업자번호'];

function krSheet() {
  const ss = book();
  let sh = ss.getSheetByName(KR_SHEET);
  if (!sh) {
    sh = ss.insertSheet(KR_SHEET);
    sh.setFrozenRows(1);
    sh.getRange('G:I').setNumberFormat('@');   // 전화번호·우편번호 앞자리 0 유지
  }
  sh.getRange(1, 1, 1, KR_HEAD.length).setValues([KR_HEAD]).setFontWeight('bold');   // 새 열(구분·상호명·사업자번호)도 제목 채움
  return sh;
}
const cell = v => { v = String(v == null ? '' : v).trim().slice(0, 200); return /^[=+\-@]/.test(v) ? "'" + v : v; };

// 주문서 저장 — 금액은 여기서 계산 (브라우저가 보낸 금액은 쓰지 않음)
function krOrder(b) {
  const b2b = b.type === 'b2b';
  let qty = parseInt(b.qty, 10) || 0;
  if (b2b) { if (qty < B2B.step || qty > B2B.max || qty % B2B.step) return { ok: false, error: 'qty' }; }
  else qty = Math.max(1, Math.min(KR_MAX, qty || 1));
  const phone = String(b.phone || '').replace(/[^0-9]/g, '');
  if (!b.name || !/^01[0-9]{8,9}$/.test(phone) || !b.zip || !b.addr1 || !b.agree || (b2b && !b.shop)) return { ok: false, error: 'invalid' };
  const orderNo = (b2b ? 'OSB' : 'OS') + Utilities.formatDate(new Date(), TZ, 'yyMMdd') + '-' + Math.random().toString(36).slice(2, 7).toUpperCase();
  const unit = b2b ? (qty >= B2B.bulkFrom ? B2B.bulkUnit : B2B.unit) : KR_PRICE;
  const price = qty * unit;
  krSheet().appendRow([
    Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd HH:mm'), orderNo, '결제 대기', qty, price,
    cell(b.name), phone, cell(b.email), cell(b.zip), cell(b.addr1), cell(b.addr2), cell(b.memo), '', '', '',
    b2b ? '도매' : '개인', cell(b2b ? b.shop : ''), cell(b2b ? b.biz : '')
  ]);
  return { ok: true, type: b2b ? 'b2b' : 'retail', orderNo: orderNo, price: price,
           goodname: (b2b ? '[도매] ' : '') + 'Glow Drop 글로우드롭 15ml x ' + qty };
}

// 페이앱 결제 알림 (feedbackurl) — 폼 형식으로 들어옴. 'SUCCESS' 를 돌려줘야 페이앱이 재전송하지 않음
function payappFeedback(p) {
  const want = prop('PAYAPP_LINKVAL');
  if (want && p.linkval !== want) return 'FAIL';
  const sh = krSheet();
  const last = sh.getLastRow();
  if (last < 2) return 'SUCCESS';
  const ids = sh.getRange(2, 2, last - 1, 1).getValues();
  let row = 0;
  for (let i = ids.length - 1; i >= 0; i--) if (ids[i][0] === p.var1) { row = i + 2; break; }
  if (!row) return 'SUCCESS';
  const st = String(p.pay_state);
  if (st === '4') {                                   // 결제 완료
    const expected = Number(sh.getRange(row, 5).getValue());
    const paid = Number(p.price);
    sh.getRange(row, 3).setValue(paid === expected ? '결제 완료' : '금액 확인 필요');
    sh.getRange(row, 13, 1, 3).setValues([[cell(p.mul_no), cell(p.pay_date || Utilities.formatDate(new Date(), TZ, 'yyyy-MM-dd HH:mm')), paid || '']]);
  } else if (['8', '9', '16', '32', '64', '70', '71'].indexOf(st) >= 0) {
    sh.getRange(row, 3).setValue('결제 취소');
  }
  return 'SUCCESS';
}

/* ===== 방문자 · 나라별 (Google Analytics 4) — 최근 28일 + 지금 접속 중(최근 30분) ===== */
function gaRows(rep, withSessions) {
  return (rep.rows || []).map(r => ({
    code: r.dimensionValues[0].value, country: r.dimensionValues[1].value,
    users: Number(r.metricValues[0].value) || 0,
    sessions: withSessions ? Number(r.metricValues[1].value) || 0 : undefined
  }));
}
function visitors() {
  const id = prop('GA4_PROPERTY_ID');
  if (!id) return { ok: false, error: 'no_ga' };
  if (typeof AnalyticsData === 'undefined') return { ok: false, error: 'no_service' };
  const property = 'properties/' + String(id).replace(/[^0-9]/g, '');
  const cache = CacheService.getScriptCache();

  let month = cache.get('visitors28');
  if (month) month = JSON.parse(month);
  else {
    const rows = gaRows(AnalyticsData.Properties.runReport({
      dateRanges: [{ startDate: '28daysAgo', endDate: 'today' }],
      dimensions: [{ name: 'countryId' }, { name: 'country' }],
      metrics: [{ name: 'activeUsers' }, { name: 'sessions' }],
      orderBys: [{ metric: { metricName: 'activeUsers' }, desc: true }],
      limit: 250
    }, property), true);
    month = { rows: rows, users: rows.reduce((a, r) => a + r.users, 0) };
    cache.put('visitors28', JSON.stringify(month), 1800);   // 30분 보관
  }

  let live = cache.get('visitorsLive');
  if (live) live = JSON.parse(live);
  else {
    let rows = [];
    try {
      rows = gaRows(AnalyticsData.Properties.runRealtimeReport({
        dimensions: [{ name: 'countryId' }, { name: 'country' }],
        metrics: [{ name: 'activeUsers' }],
        limit: 250
      }, property), false);
    } catch (e) { console.error('realtime', e); }
    live = { rows: rows, users: rows.reduce((a, r) => a + r.users, 0) };
    cache.put('visitorsLive', JSON.stringify(live), 60);   // 1분 보관
  }
  return { ok: true, days: 28, rows: month.rows, users: month.users, live: live, at: Date.now() };
}

const SCRIPT_VERSION = '2026-09-27e';
function doGet() { let sheet = ''; try { sheet = book().getName(); } catch (e) { sheet = ''; }
  return out({ ok: true, service: 'olive-skin', version: SCRIPT_VERSION, adminKey: !!prop('ADMIN_KEY'), sheet: sheet }); }
const CHECKOUT_ACTIONS = ['create', 'patch', 'capture', 'krOrder'];
function doPost(e) {
  // 페이앱 결제 알림은 JSON 이 아니라 폼 형식
  if (e.parameter && e.parameter.mul_no) {
    const lock = LockService.getScriptLock();
    lock.waitLock(20000);
    try { return ContentService.createTextOutput(payappFeedback(e.parameter)); }
    catch (x) { console.error(x); return ContentService.createTextOutput('FAIL'); }
    finally { lock.releaseLock(); }
  }
  let b;
  try { b = JSON.parse(e.postData.contents); } catch (x) { return out({ ok: false, error: 'bad_request' }); }

  // 결제 (비밀번호 없이 — 금액은 여기서 계산)
  if (CHECKOUT_ACTIONS.indexOf(b.action) >= 0) {
    try {
      if (b.action === 'create') return out(createOrder(b));
      if (b.action === 'patch') return out(patchOrder(b));
      const lock = LockService.getScriptLock();
      lock.waitLock(20000);
      try { return out(b.action === 'capture' ? captureOrder(b) : krOrder(b)); } finally { lock.releaseLock(); }
    } catch (x) { console.error(x); return out({ ok: false, error: 'server' }); }
  }

  // 관리자 (비밀번호 필요)
  // 스크립트 속성에 ADMIN_KEY 가 없으면 '비밀번호 틀림'이 아니라 설정이 빠진 것 — 따로 알려 줌
  if (!prop('ADMIN_KEY')) return out({ ok: false, error: 'no_admin_key' });
  if (!authed(b.key)) return out({ ok: false, error: 'auth' });
  if (b.action === 'visitors') {
    try { return out(visitors()); } catch (x) { return out({ ok: false, error: 'ga', detail: String(x && x.message || x) }); }
  }
  const lock = LockService.getScriptLock();
  try { lock.waitLock(10000); } catch (x) { return out({ ok: false, error: 'busy' }); }
  try {
    switch (b.action) {
      case 'list': break;
      case 'save': saveRow(ORDERS, b.order || {}); break;
      case 'delete': deleteRow(ORDERS, b.id); break;
      case 'settings': saveSettings(b.settings || {}); break;
      case 'saveVendor': saveRow(VENDORS, b.vendor || {}); break;
      case 'deleteVendor': deleteRow(VENDORS, b.id); break;
      case 'mkt': return out({ ok: true, mkt: mktRead() });
      case 'mktSave': mktWrite(b.mkt); return out({ ok: true, mkt: mktRead() });
      default: return out({ ok: false, error: 'bad_action' });
    }
    return out(Object.assign({ ok: true }, readAll()));
  } catch (x) {
    return out({ ok: false, error: 'server', detail: String(x && x.message || x) });
  } finally { lock.releaseLock(); }
}
